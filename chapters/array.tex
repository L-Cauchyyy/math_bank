\chapter{1}{数列}

\subchapter{1}{基础知识}

\begin{definition}[等差数列]
如果一个数列从第二项起，每一项与它前一项的差都等于同一个常数 $d$，那么这个数列叫做等差数列，
$d$ 叫做公差。
\end{definition}

\begin{theorem}[等差数列通项公式与前 $n$ 项和]
若 $\{a_n\}$ 是等差数列，则
\[
a_n = a_1+(n-1)d, \qquad
S_n = \frac{n(a_1+a_n)}{2} = na_1+\frac{n(n-1)}{2}d .
\]
\end{theorem}
