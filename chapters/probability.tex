\chapter{5}{概率统计}

\begin{exercise}
  设$A,B$是一个随机试验中的两个事件，$P(A|B)+P(B|A)=1,\ P(A\cup B)=\dfrac{3}{4}$，则
\begin{enumerate}[label=\Alph*.,itemsep=4pt]
    \item 事件$A,B$相互独立
    \item 若$P(AB)=\dfrac{1}{8}$，则$|P(A)-P(B)|=\dfrac{\sqrt{21}}{8}$
    \item $P(AB)\leqslant P(\overline{A}\overline{B})$
    \item 若$P(A|\overline{B})=P(\overline{A}|B)$，则必有$P(A)=P(B)$
\end{enumerate}
\end{exercise}
